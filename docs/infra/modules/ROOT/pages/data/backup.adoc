=== CloudNativePG Backup-Konfiguration (geplant)

[source,yaml]
----
apiVersion: postgresql.cnpg.io/v1
kind: Cluster
metadata:
  name: postgres-cluster
spec:
  # ... bestehende Konfiguration ...
  backup:
    barmanObjectStore:
      destinationPath: s3://tomsblog-backups/postgres/
      endpointURL: https://garage.netcup.example.com
      s3Credentials:
        accessKeyId:
          name: backup-s3-credentials
          key: ACCESS_KEY_ID
        secretAccessKey:
          name: backup-s3-credentials
          key: SECRET_ACCESS_KEY
      wal:
        compression: gzip
    retentionPolicy: "30d"
----
